EXERCICE 1

fonction decalage_gauche(tab, n) {
    si (n <= 1) {
        retour tab
    }
    premier = tab[0]
    pour i allant de 0 a n - 2 {
        tab[i] = tab[i + 1]
    }
    tab[n - 1] = premier
    retour tab
}

Complexite temporelle : O(n)
Complexite spatiale : O(1)


EXERCICE 2

fonction produit(X, m, Y, n) {
    taille_max = m + n
    R = {}
    pour k allant de 0 a taille_max - 1 {
        R[k] = 0
    }
    pour i allant de 0 a m - 1 {
        pour j allant de 0 a n - 1 {
            R[i + j] = R[i + j] + (X[i] * Y[j])
        }
    }
    retenue = 0
    pour k allant de 0 a taille_max - 1 {
        somme = R[k] + retenue
        R[k] = somme % 10
        retenue = somme / 10
    }
    retour R
}

Complexite temporelle : O(m * n)
Complexite spatiale : O(m + n)


EXERCICE 3

fonction facteurs_premiers(n) {
    facteurs = {}
    taille = 0
    m = n
    d = 2
    tant que (d * d <= m) {
        si (m % d == 0) {
            facteurs[taille] = d
            taille = taille + 1
            m = m / d
        } sinon {
            d = d + 1
        }
    }
    si (m > 1) {
        facteurs[taille] = m
        taille = taille + 1
    }
    retour facteurs
}

Complexite temporelle : O(sqrt(n)) au pire cas
Complexite spatiale : O(log n)


EXERCICE 4

fonction sinus_ordre_n(x, n) {
    somme = x
    terme = x
    pour k allant de 2 a n {
        terme = - terme * x * x / ((2 * k - 2) * (2 * k - 1))
        somme = somme + terme
    }
    retour somme
}

Complexite temporelle : O(n)
Complexite spatiale : O(1)


EXERCICE 5

fonction est_palindrome(n) {
    si (n < 0) {
        retour faux
    }
    si (n < 10) {
        retour vrai
    }
    si (n % 10 == 0) {
        retour faux
    }
    x = n
    y = 0
    tant que (x > 0) {
        y = (y * 10) + (x % 10)
        x = x / 10
    }
    si (y == n) {
        retour vrai
    } sinon {
        retour faux
    }
}

Complexite temporelle : O(log10(n))
Complexite spatiale : O(1)
